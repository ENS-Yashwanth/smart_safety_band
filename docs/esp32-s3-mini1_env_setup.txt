1. Update Ubuntu
	sudo apt update
	sudo apt upgrade
2. Install the required tools
	sudo apt install git wget flex bison gperf python3 python3-pip \
		python3-venv cmake ninja-build ccache libffi-dev libssl-dev \
		dfu-util libusb-1.0-0
3. Verify the version
	cmake --version
	python3 --version
	git --version
	ninja --version
4. Create a Workspace, Install ESP-IDF
	mkdir ~/esp
	cd ~/esp
	git clone --recursive https://github.com/espressif/esp-idf.git
	cd esp-idf
	./install.sh esp32s3
This downloads - Xtensa Compiler, OpenOCD, Python Packages, Flash Tools
5. Configure Environment
	source ~/esp/esp-idf/export.sh
Every new terminal needs the environment variables
To make this automatic 
	echo "source ~/esp/esp-idf/export.sh" >> ~/.bashrc
	source ~/.bashrc
6. Verify installation
	idf.py --version
	xtensa-esp32s3-elf-gcc --version
7. Set the target
	idf.py set-target esp32s3
8. Configure 
	idf.py meuconfig
	idf.py reconfigure
9. Build
	idf.py build
10. Flash
	idf.py -p /dev/ttyACM0 flash
	idf.py flash
11. Monitor
	idf.py -p /dev/ttyACM0 monitor
	idf.py monitor
12. Cleanup all
	idf.py fullclean
13. Execute all at once
	idf.py fullclean reconfigure build flash monitor
14. Erase the flash program 
	idf.py -p /dev/ttyACM0 erase-flash
	idf.py erase-flash
